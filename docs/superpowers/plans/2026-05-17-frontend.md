# ParkSmart Frontend (Plan 4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the ParkSmart React single-page app — a mobile-first live parking map for Chatswood CBD with a time-scrubber for predictions, a street zone-rules search, and an LLM chat assistant.

**Architecture:** A Vite + React + TypeScript SPA. A full-bleed Leaflet map is the base layer; a search field, a time scrubber, and an assistant panel float over it. One CSS breakpoint re-flows the assistant between a mobile bottom sheet and a 380px desktop drawer. A single `viewTime` state switches the map between live polling and predicted occupancy. State: React Context for chat, local React state for the map. The frontend has no automated test suite (project decision) — correctness rests on TypeScript, a typed API client, and the manual QA pass in the final task. One small backend endpoint (`GET /api/zones/streets`) is added for street autocomplete.

**Tech Stack:** React 18/19, Vite, TypeScript, Tailwind CSS, Leaflet + react-leaflet, lucide-react (icons). Backend: FastAPI, pytest (for the one backend task).

**Spec:** `docs/superpowers/specs/2026-05-17-parksmart-frontend-design.md`

**Commits:** Per project convention (`CLAUDE.md`), this plan contains no `git commit` steps — the user commits manually between tasks. No `Co-Authored-By` footer.

```python
def list_streets(db_path: Path) -> list[str]:
    """All known street names — distinct geocoded streets from parking_signs
    plus any streets defined in seed_zones.json — case-insensitively deduped
    and sorted. Powers the frontend street-search autocomplete."""
    with get_connection(db_path) as con:
        rows = con.execute(
            "SELECT DISTINCT street FROM parking_signs "
            "WHERE street IS NOT NULL AND street != ''"
        ).fetchall()

    streets: dict[str, str] = {r["street"].lower(): r["street"] for r in rows}
    for entry in _load_seed().values():
        name = entry["street"]
        streets.setdefault(name.lower(), name)

    return sorted(streets.values(), key=str.lower)
```

```ts
import type {
  OccupancyResponse,
  PredictionResult,
  ZoneResponse,
} from "./types";

export class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message);
    this.name = "ApiError";
  }
}

async function getJSON<T>(url: string): Promise<T> {
  const resp = await fetch(url);
  if (!resp.ok) {
    throw new ApiError(resp.status, `Request failed: ${resp.status}`);
  }
  return (await resp.json()) as T;
}

export function fetchOccupancy(): Promise<OccupancyResponse[]> {
  return getJSON<OccupancyResponse[]>("/api/occupancy");
}

export function fetchPrediction(
  location: string,
  targetDatetime: string,
): Promise<PredictionResult> {
  const params = new URLSearchParams({
    location,
    target_datetime: targetDatetime,
  });
  return getJSON<PredictionResult>(`/api/predict?${params.toString()}`);
}

export function fetchZone(street: string): Promise<ZoneResponse> {
  return getJSON<ZoneResponse>(`/api/zones?street=${encodeURIComponent(street)}`);
}

export function fetchStreets(): Promise<string[]> {
  return getJSON<string[]>("/api/zones/streets");
}
```

```ts
import type {
  MarkerData,
  OccupancyResponse,
  PredictionResult,
} from "../types";

export type ColourBand = "green" | "amber" | "red";

export const BAND_HEX: Record<ColourBand, string> = {
  green: "#3fa34d",
  amber: "#e6a020",
  red: "#d6453d",
};

export function colourBand(pct: number): ColourBand {
  if (pct < 0.7) return "green";
  if (pct < 0.9) return "amber";
  return "red";
}

export interface Badge {
  label: string;
  tone: "live" | "estimated" | "predicted";
}

export function badgeFor(m: MarkerData): Badge {
  if (m.mode === "live") {
    return m.source === "tfnsw"
      ? { label: "Live", tone: "live" }
      : { label: "Estimated", tone: "estimated" };
  }
  // predicted
  if (m.modelVersion === "xgboost-v1") {
    const conf = Math.round((m.confidence ?? 0) * 100);
    return { label: `Predicted · ${conf}% confidence`, tone: "predicted" };
  }
  return { label: "Estimated · typical patterns", tone: "estimated" };
}
```

```ts
import { useEffect, useState } from "react";

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(
    () => window.matchMedia(query).matches,
  );

  useEffect(() => {
    const mql = window.matchMedia(query);
    const handler = (e: MediaQueryListEvent) => setMatches(e.matches);
    mql.addEventListener("change", handler);
    setMatches(mql.matches);
    return () => mql.removeEventListener("change", handler);
  }, [query]);

  return matches;
}
```

- [ ] **Step 1: Write `frontend/src/hooks/useOccupancy.ts`**

Polls every 5 minutes. On error it keeps the last good data and exposes an error string.

```ts
import { useCallback, useEffect, useState } from "react";
import { fetchOccupancy } from "../api";
import type { OccupancyResponse } from "../types";

const REFRESH_MS = 5 * 60 * 1000;

export function useOccupancy() {
  const [occupancy, setOccupancy] = useState<OccupancyResponse[] | null>(null);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async () => {
    try {
      const fresh = await fetchOccupancy();
      setOccupancy(fresh);
      setError(null);
    } catch {
      setError("Could not refresh parking data — showing the last known values.");
    }
  }, []);

  useEffect(() => {
    void load();
    const id = window.setInterval(() => void load(), REFRESH_MS);
    return () => window.clearInterval(id);
  }, [load]);

  return { occupancy, error };
}
```

```tsx
import { useEffect, useRef } from "react";
import { MapContainer, TileLayer, useMap } from "react-leaflet";
import L from "leaflet";
import type { MarkerData } from "../../types";
import { CarParkMarker } from "./CarParkMarker";

function FitToMarkers({ markers }: { markers: MarkerData[] }) {
  const map = useMap();
  const fitted = useRef(false);

  useEffect(() => {
    if (fitted.current || markers.length === 0) return;
    const bounds = L.latLngBounds(markers.map((m) => [m.lat, m.lon]));
    map.fitBounds(bounds, { padding: [60, 60] });
    fitted.current = true;
  }, [markers, map]);

  return null;
}

interface Props {
  markers: MarkerData[];
  selectedId: string | null;
  showPopups: boolean;
  onSelect: (id: string) => void;
}

export function MapView({ markers, selectedId, showPopups, onSelect }: Props) {
  return (
    <MapContainer
      center={[-33.797, 151.183]}
      zoom={14}
      zoomControl={false}
      className="absolute inset-0 z-0"
    >
      <TileLayer
        url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        attribution='&copy; OpenStreetMap contributors'
      />
      <FitToMarkers markers={markers} />
      {markers.map((m) => (
        <CarParkMarker

          marker={m}
```

- [ ] **Step 1: Write `frontend/src/hooks/usePredictions.ts`**

Fans out one `/api/predict` call per car park (by name) with `Promise.allSettled`, so one failure does not break the rest.

```ts
import { useEffect, useState } from "react";
import { fetchPrediction } from "../api";
import type { OccupancyResponse, PredictionResult } from "../types";

export function usePredictions(
  carParks: OccupancyResponse[] | null,
  targetTime: Date | null,
) {
  const [predictions, setPredictions] = useState<
    Map<string, PredictionResult>
  >(new Map());
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (!targetTime || !carParks || carParks.length === 0) {
      setPredictions(new Map());
      return;
    }
    let cancelled = false;
    setLoading(true);
    const iso = targetTime.toISOString();

    Promise.allSettled(
      carParks.map((cp) => fetchPrediction(cp.name, iso)),
    ).then((results) => {
      if (cancelled) return;
      const map = new Map<string, PredictionResult>();
      results.forEach((r, i) => {
        if (r.status === "fulfilled") {
          map.set(carParks[i].car_park_id, r.value);
        }
      });
      setPredictions(map);
      setLoading(false);
    });

    return () => {
      cancelled = true;
    };
  }, [carParks, targetTime]);

  return { predictions, loading };
}
```

- [ ] **Step 2: Write `frontend/src/components/Scrubber/TimeScrubber.tsx`**

The slider spans 0–168 hours (0 = now = live mode). The time chip is a `datetime-local` input for exact entry. Both bind to one `viewTime`. The reference "now" is captured once on mount.

```tsx
import { useMemo } from "react";

const MAX_HOURS = 168; // 7 days
const HOUR_MS = 60 * 60 * 1000;

function floorToHour(d: Date): Date {
  const r = new Date(d);
  r.setMinutes(0, 0, 0);
  return r;
}

/** Format a Date for the value of an <input type="datetime-local">. */
function toLocalInputValue(d: Date): string {
  const pad = (n: number) => String(n).padStart(2, "0");
  return (
    `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}` +
    `T${pad(d.getHours())}:${pad(d.getMinutes())}`
  );
}

interface Props {
  viewTime: Date | null;
  onChange: (t: Date | null) => void;
}

export function TimeScrubber({ viewTime, onChange }: Props) {
  const now = useMemo(() => floorToHour(new Date()), []);
  const maxTime = useMemo(
    () => new Date(now.getTime() + MAX_HOURS * HOUR_MS),
    [now],
  );

  const hours = viewTime
    ? Math.round((viewTime.getTime() - now.getTime()) / HOUR_MS)
    : 0;

  const handleSlider = (e: React.ChangeEvent<HTMLInputElement>) => {
    const h = Number(e.target.value);
    onChange(h <= 0 ? null : new Date(now.getTime() + h * HOUR_MS));
  };

  const handleTyped = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (!e.target.value) {
      onChange(null);
      return;
    }
    const picked = floorToHour(new Date(e.target.value));
    const clamped = new Date(
      Math.min(Math.max(picked.getTime(), now.getTime()), maxTime.getTime()),
    );
    onChange(clamped.getTime() <= now.getTime() ? null : clamped);
  };

  return (
    <div className="absolute bottom-4 left-1/2 z-[1000] w-[min(560px,90vw)] -translate-x-1/2 rounded-full bg-white px-4 py-2.5 shadow-lg">
      <div className="flex items-center gap-3 text-xs text-slate-600">
        <span className="font-semibold">Now</span>
        <input
          type="range"
          min={0}
          max={MAX_HOURS}
          value={hours}
          onChange={handleSlider}
          className="flex-1 accent-blue-600"
          aria-label="Prediction time"
        />
        <span className="font-semibold">+7d</span>
        <input
          type="datetime-local"
          value={toLocalInputValue(viewTime ?? now)}
          min={toLocalInputValue(now)}
          max={toLocalInputValue(maxTime)}
          onChange={handleTyped}
          className="rounded bg-slate-800 px-2 py-1 text-[11px] font-semibold text-white"
          aria-label="Exact prediction time"
        />
      </div>
      <p className="mt-1 text-center text-[11px] text-slate-500">
        {viewTime
          ? `Predicted · ${viewTime.toLocaleString([], {
              weekday: "short",
              hour: "numeric",
              minute: "2-digit",
            })}`
          : "Live · now"}
      </p>
    </div>
  );
}
```

```ts
import { useCallback, useEffect, useState } from "react";
import { ApiError, fetchStreets, fetchZone } from "../api";
import type { ZoneResponse } from "../types";

type Status = "idle" | "loading" | "loaded" | "not_found" | "error";

export function useZones() {
  const [streets, setStreets] = useState<string[]>([]);
  const [zone, setZone] = useState<ZoneResponse | null>(null);
  const [status, setStatus] = useState<Status>("idle");

  useEffect(() => {
    fetchStreets()
      .then(setStreets)
      .catch(() => setStreets([]));
  }, []);

  const lookup = useCallback(async (street: string) => {
    if (!street.trim()) return;
    setStatus("loading");
    setZone(null);
    try {
      const result = await fetchZone(street);
      setZone(result);
      setStatus("loaded");
    } catch (e) {
      setStatus(e instanceof ApiError && e.status === 404 ? "not_found" : "error");
    }
  }, []);

  const clear = useCallback(() => {
    setZone(null);
    setStatus("idle");
  }, []);

  return { streets, zone, status, lookup, clear };
}
```

- [ ] **Step 1: Write `frontend/src/hooks/useChat.ts`**

`/api/chat` is a POST returning an SSE stream, so this uses `fetch` + a `ReadableStream` reader (the browser `EventSource` is GET-only). It splits the body on `\n\n` and parses each `data:` payload.

```ts
import { useCallback, useState } from "react";
import type { ChatEvent, ChatMessage } from "../types";

type Status = "idle" | "streaming" | "error";

export function useChat() {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [tools, setTools] = useState<string[]>([]);
  const [status, setStatus] = useState<Status>("idle");

  const send = useCallback(
    async (text: string) => {
      if (!text.trim() || status === "streaming") return;

      const history = messages.map((m) => ({
        role: m.role,
        content: m.content,
      }));
      setMessages((prev) => [
        ...prev,
        { role: "user", content: text },
        { role: "assistant", content: "" },
      ]);
      setTools([]);
      setStatus("streaming");

      const appendToAssistant = (chunk: string) => {
        setMessages((prev) => {
          const next = [...prev];
          const last = next[next.length - 1];
          next[next.length - 1] = {
            ...last,
            content: last.content + chunk,
          };
          return next;
        });
      };

      try {
        const resp = await fetch("/api/chat", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ message: text, history }),
        });
        if (!resp.ok || !resp.body) {
          throw new Error(`Chat failed (${resp.status})`);
        }

        const reader = resp.body.getReader();
        const decoder = new TextDecoder();
        let buffer = "";

        for (;;) {
          const { done, value } = await reader.read();
          if (done) break;
          buffer += decoder.decode(value, { stream: true });
          const parts = buffer.split("\n\n");
          buffer = parts.pop() ?? "";
          for (const part of parts) {
            const line = part.trim();
            if (!line.startsWith("data:")) continue;
            const payload = line.slice(5).trim();
            if (!payload) continue;
            const event = JSON.parse(payload) as ChatEvent;
            if (event.type === "text") appendToAssistant(event.content);
            else if (event.type === "tool_call")
              setTools((prev) => [...prev, event.tool]);
          }
        }
        setStatus("idle");
      } catch {
        appendToAssistant(
          "\n\n⚠️ Sorry — I couldn't reach the assistant. Please try again.",
        );
        setStatus("error");
      }
    },
    [messages, status],
  );

  return { messages, tools, status, send };
}
```

```tsx
import { useState } from "react";
import { MapView } from "./components/Map/MapView";
import { TimeScrubber } from "./components/Scrubber/TimeScrubber";
import { CarParkSheet } from "./components/Map/CarParkSheet";
import { SearchField } from "./components/Zones/SearchField";
import { AssistantPanel } from "./components/Assistant/AssistantPanel";
import { ChatProvider } from "./context/ChatContext";
import { useOccupancy } from "./hooks/useOccupancy";
import { usePredictions } from "./hooks/usePredictions";
import { useMediaQuery } from "./hooks/useMediaQuery";
import { liveToMarker, predictionToMarker } from "./lib/occupancy";
import type { MarkerData } from "./types";

export default function App() {
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [viewTime, setViewTime] = useState<Date | null>(null);
  const { occupancy, error } = useOccupancy();
  const { predictions } = usePredictions(occupancy, viewTime);
  const isDesktop = useMediaQuery("(min-width: 768px)");

  const markers: MarkerData[] = (occupancy ?? []).map((o) => {
    const p = predictions.get(o.car_park_id);
    return viewTime && p ? predictionToMarker(o, p) : liveToMarker(o);
  });

  const selected = markers.find((m) => m.id === selectedId) ?? null;

  return (
    <ChatProvider>
      <div className="relative h-screen w-screen overflow-hidden">
        <MapView
          markers={markers}
          selectedId={selectedId}
          showPopups={isDesktop}
          onSelect={setSelectedId}
        />
        <SearchField />
        <TimeScrubber viewTime={viewTime} onChange={setViewTime} />
        <AssistantPanel isDesktop={isDesktop} />
        {!isDesktop && (
          <CarParkSheet marker={selected} onClose={() => setSelectedId(null)} />
        )}
        {error && (
          <div className="absolute bottom-20 left-1/2 z-[1100] -translate-x-1/2 rounded-lg bg-amber-100 px-4 py-2 text-sm text-amber-900 shadow">
            {error}
          </div>
        )}
      </div>
    </ChatProvider>
  );
}
```
